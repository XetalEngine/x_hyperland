cmake_minimum_required(VERSION 3.16)
project(FusionFiles VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus Concurrent)
find_package(KF6KIO 6.13 REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6JobWidgets REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6WindowSystem REQUIRED)
add_library(fusion-files-ui STATIC core.cpp drives.cpp drive_properties.cpp pathbar.cpp browser.cpp window.cpp picker.cpp transfers.cpp transfer_access.cpp applications.cpp removal.cpp trash_animation.cpp)
target_link_libraries(fusion-files-ui PUBLIC Qt6::Widgets Qt6::DBus Qt6::Concurrent
    KF6::KIOCore KF6::KIOGui KF6::KIOWidgets KF6::KIOFileWidgets KF6::Service KF6::JobWidgets KF6::CoreAddons KF6::WindowSystem)
target_include_directories(fusion-files-ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(xetal-files main.cpp)
target_link_libraries(xetal-files PRIVATE fusion-files-ui)
option(BUILD_TESTING "Build isolated Fusion Files checks" OFF)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    enable_testing()
    add_executable(files-tests tests.cpp)
    target_link_libraries(files-tests PRIVATE fusion-files-ui Qt6::Test)
    add_executable(picker-tests picker_tests.cpp)
    target_link_libraries(picker-tests PRIVATE fusion-files-ui Qt6::Test)
    add_test(NAME fusion-picker COMMAND picker-tests)
    set_tests_properties(fusion-picker PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=")
    add_executable(files-drag-probe drag_probe.cpp)
    target_link_libraries(files-drag-probe PRIVATE fusion-files-ui)
    add_test(NAME fusion-files COMMAND files-tests)
    set_tests_properties(fusion-files PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=")
endif()
